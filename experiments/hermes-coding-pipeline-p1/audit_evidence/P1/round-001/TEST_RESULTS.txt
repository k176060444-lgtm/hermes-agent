def test_windows_cross_process_lock_fails_closed(tmp_path):
    lock = tmp_path / 'independent-process.lock'
    holder_code = ('import sys; from pathlib import Path; from p1.state import exclusive_lock; '
                   'cm=exclusive_lock(Path(sys.argv[1])); cm.__enter__(); '
                   'print("HELD", flush=True); sys.stdin.readline(); cm.__exit__(None,None,None)')
    contender_code = ('import sys; from pathlib import Path; from p1.state import exclusive_lock,StateError; '
                      'cm=exclusive_lock(Path(sys.argv[1])); '
                      '\ntry:\n cm.__enter__()\nexcept StateError:\n print("BLOCKED")\nelse:\n print("ACQUIRED")\n cm.__exit__(None,None,None)')
    holder = subprocess.Popen([sys.executable, '-B', '-c', holder_code, str(lock)],
                              stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                              text=True, encoding='utf-8')
    try:
        held = holder.stdout.readline().strip()
        probe = subprocess.run([sys.executable, '-B', '-c', contender_code, str(lock)],
                               capture_output=True, text=True, encoding='utf-8', timeout=10)
        print('holder:', held, '; competing process:', probe.stdout.strip(), '; exit:', probe.returncode)
        assert held == 'HELD' and probe.returncode == 0 and probe.stdout.strip() == 'BLOCKED'
    finally:
        holder.communicate(input='\n', timeout=10)
